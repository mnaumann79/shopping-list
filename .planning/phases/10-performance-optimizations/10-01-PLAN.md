   ```typescript
   }, (prev, next) => {
     // Return true = props are equal = SKIP re-render
     if (prev.item.id !== next.item.id) return false;
     if (prev.item.name !== next.item.name) return false;
     if (prev.item.quantity !== next.item.quantity) return false;
     if (prev.item.note !== next.item.note) return false;
     if (prev.item.categoryId !== next.item.categoryId) return false;
     if (prev.item.isChecked !== next.item.isChecked) return false;
     if (!prev.item.storeIds || !next.item.storeIds) return prev.item.storeIds === next.item.storeIds;
     if (prev.item.storeIds.length !== next.item.storeIds.length) return false;
     if (!prev.item.storeIds.every((id, i) => id === next.item.storeIds![i])) return false;
     if (prev.stores !== next.stores) return false;
     if (prev.categories !== next.categories) return false;
     return true;
   });
   ```

   The comparator also checks `stores` and `categories` reference equality -- these are stable useState references that only change when the data actually changes, so reference comparison is sufficient and ensures the edit dialog dropdowns stay current.
