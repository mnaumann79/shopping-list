```typescript
import { auth } from "@/lib/auth";
import { db } from "@/db";
import { householdJoinLinks } from "@/db/schema/households";
import { headers } from "next/headers";
import { NextRequest } from "next/server";
import { randomBytes } from "crypto";
import { eq } from "drizzle-orm";

export async function POST(req: NextRequest) {
  const session = await auth.api.getSession({ headers: await headers() });
  if (!session) {
    return Response.json({ error: "Unauthorized" }, { status: 401 });
  }

  const body = await req.json().catch(() => null);
  if (!body?.organizationId) {
    return Response.json({ error: "organizationId is required" }, { status: 400 });
  }

  const token = randomBytes(32).toString("hex"); // 64 hex chars
  const expiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000); // 7 days

  // Regenerate: delete existing link(s) for this org first
  await db
    .delete(householdJoinLinks)
    .where(eq(householdJoinLinks.organizationId, body.organizationId));

  await db.insert(householdJoinLinks).values({
    token,
    organizationId: body.organizationId,
    createdById: session.user.id,
    expiresAt,
  });

  const baseUrl = process.env.BETTER_AUTH_URL ?? process.env.NEXT_PUBLIC_APP_URL ?? "http://localhost:3000";
  const url = `${baseUrl}/join?token=${token}`;

  return Response.json({ url }, { status: 200 });
}
```

```typescript
import { auth } from "@/lib/auth";
import { db } from "@/db";
import { householdJoinLinks } from "@/db/schema/households";
import { headers } from "next/headers";
import { NextRequest } from "next/server";
import { eq } from "drizzle-orm";

export async function POST(req: NextRequest) {
  const session = await auth.api.getSession({ headers: await headers() });
  if (!session) {
    return Response.json({ error: "Unauthorized" }, { status: 401 });
  }

  const body = await req.json().catch(() => null);
  if (!body?.token) {
    return Response.json({ error: "token is required" }, { status: 400 });
  }

  const [link] = await db
    .select()
    .from(householdJoinLinks)
    .where(eq(householdJoinLinks.token, body.token))
    .limit(1);

  if (!link) {
    return Response.json({ error: "Invalid join link" }, { status: 404 });
  }

  if (link.expiresAt < new Date()) {
    return Response.json({ error: "This join link has expired" }, { status: 410 });
  }

  // Add member to household — do NOT delete token (multi-use)
  await auth.api.addMember({
    body: {
      userId: session.user.id,
      organizationId: link.organizationId,
      role: "member",
    },
    headers: await headers(),
  });

  return Response.json({ success: true, organizationId: link.organizationId }, { status: 200 });
}
```

```tsx
"use client";
import { useEffect, useState } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { authClient } from "@/lib/auth-client";
import { Card, CardHeader, CardTitle, CardDescription, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

export default function JoinPage() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const token = searchParams.get("token");
  const [status, setStatus] = useState<"loading" | "success" | "error" | "expired">("loading");
  const [errorMessage, setErrorMessage] = useState("");

  useEffect(() => {
    if (!token) {
      setStatus("error");
      setErrorMessage("Invalid join link — no token found.");
      return;
    }

    async function handleJoin() {
      const sessionResult = await authClient.getSession();

      if (!sessionResult.data) {
        // Not logged in: preserve token, redirect to sign-up
        sessionStorage.setItem("pendingJoinToken", token!);
        router.push(`/sign-up?joinToken=${token}`);
        return;
      }

      // Logged in: auto-join via API
      try {
        const res = await fetch("/api/households/join", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token }),
        });

        if (res.status === 410) {
          setStatus("expired");
          setErrorMessage("This join link has expired. Ask the household admin to generate a new one.");
          return;
        }

        if (!res.ok) {
          const data = await res.json().catch(() => ({}));
          setStatus("error");
          setErrorMessage(data.error ?? "Failed to join household");
          return;
        }

        const data = await res.json();
        // Set the joined household as active
        await authClient.organization.setActive({ organizationId: data.organizationId });
        setStatus("success");
        setTimeout(() => {
          router.push("/dashboard");
          router.refresh();
        }, 1500);
      } catch {
        setStatus("error");
        setErrorMessage("An unexpected error occurred.");
      }
    }

    handleJoin();
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [token]);

  if (status === "loading") {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <p className="text-muted-foreground">Joining household...</p>
      </div>
    );
  }

  if (status === "success") {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <Card className="w-full max-w-sm">
          <CardHeader>
            <CardTitle>Joined!</CardTitle>
            <CardDescription>You have joined the household. Redirecting to your dashboard...</CardDescription>
          </CardHeader>
        </Card>
      </div>
    );
  }

  if (status === "expired") {
    return (
      <div className="min-h-screen flex items-center justify-center px-4">
        <Card className="w-full max-w-sm">
          <CardHeader>
            <CardTitle>Link expired</CardTitle>
            <CardDescription>{errorMessage}</CardDescription>
          </CardHeader>
          <CardContent>
            <Button variant="outline" onClick={() => router.push("/sign-in")} className="w-full">
              Go to sign in
            </Button>
          </CardContent>
        </Card>
      </div>
    );
  }

  return (
    <div className="min-h-screen flex items-center justify-center px-4">
      <Card className="w-full max-w-sm">
        <CardHeader>
          <CardTitle>Unable to join</CardTitle>
          <CardDescription>{errorMessage}</CardDescription>
        </CardHeader>
        <CardContent>
          <Button variant="outline" onClick={() => router.push("/sign-in")} className="w-full">
            Go to sign in
          </Button>
        </CardContent>
      </Card>
    </div>
  );
}
```
