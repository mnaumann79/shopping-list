Create `e2e/auth.setup.ts`:
```typescript
import { test as setup, expect } from "@playwright/test";
import path from "path";

const authFile = path.join(__dirname, "../playwright/.auth/user.json");

setup("create authenticated user with household", async ({ page }) => {
  await page.goto("/sign-up");
  await page.getByLabel("Email").fill(process.env.TEST_EMAIL ?? "test@example.com");
  await page.getByLabel("Password").fill(process.env.TEST_PASSWORD ?? "TestPassword123!");
  await page.getByLabel("Name").fill("Test User");
  await page.getByRole("button", { name: "Sign up" }).click();
  // Onboarding: create household
  await page.waitForURL("/onboarding");
  await page.getByLabel("Household name").fill("Test Household");
  await page.getByRole("button", { name: /create/i }).click();
  await page.waitForURL("/dashboard");
  await page.context().storageState({ path: authFile });
});
```
