Create `src/lib/auth.ts`. This is a SERVER-ONLY file — never import in client components.

```typescript
import { betterAuth } from "better-auth";
import { drizzleAdapter } from "better-auth/adapters/drizzle";
import { organization } from "better-auth/plugins";
import { db } from "@/db";
import * as schema from "@/db/schema";
import { render } from "@react-email/components";

export const auth = betterAuth({
  database: drizzleAdapter(db, {
    provider: "pg",
    schema: {
      user: schema.user,
      session: schema.session,
      account: schema.account,
      verification: schema.verification,
    },
  }),
  emailAndPassword: {
    enabled: true,
  },
  session: {
    expiresIn: 60 * 60 * 24 * 30,       // 30 days
    updateAge: 60 * 60 * 24,             // extend daily on activity
    cookieCache: { enabled: true, maxAge: 5 * 60 },
  },
  plugins: [
    organization({
      schema: {
        organization: schema.organization,
        member: schema.member,
        invitation: schema.invitation,
      },
      invitationExpiresIn: 60 * 60 * 24 * 7,  // 7 days — overrides 48h default (Pitfall 5)
      async sendInvitationEmail(data) {
        // Dynamic import to keep Resend out of the initial bundle
        const { Resend } = await import("resend");
        // Import template dynamically — defined in Plan 05
        const { InviteEmailTemplate } = await import("@/lib/email/invite-template");
        const resend = new Resend(process.env.RESEND_API_KEY!);
        await resend.emails.send({
          from: process.env.RESEND_FROM_EMAIL ?? "noreply@example.com",
          to: data.email,
          subject: `You're invited to join ${data.organization.name}`,
          react: InviteEmailTemplate({
            inviterName: data.inviter.user.name,
            householdName: data.organization.name,
            inviteUrl: data.url,
          }),
        });
      },
    }),
  ],
});

export type Session = typeof auth.$Infer.Session;
```

NOTE: The `sendInvitationEmail` references `@/lib/email/invite-template` which is created in Plan 05. The dynamic import means this will not fail at module-load time even before that file exists — it only runs when an invitation is actually sent.

Create `middleware.ts` at project root:
```typescript
import { NextRequest, NextResponse } from "next/server";
import { getSessionCookie } from "better-auth/cookies";

// Routes that require authentication
const PROTECTED_PREFIXES = ["/dashboard", "/household", "/settings"];
// Auth routes (redirect away if already logged in)
const AUTH_ROUTES = ["/sign-in", "/sign-up"];
// Onboarding route — allow even without active household
const ONBOARDING_ROUTE = "/onboarding";

export function middleware(request: NextRequest) {
  const session = getSessionCookie(request);
  const { pathname } = request.nextUrl;

  const isProtected = PROTECTED_PREFIXES.some((p) => pathname.startsWith(p));
  const isAuthRoute = AUTH_ROUTES.some((p) => pathname.startsWith(p));

  // Not authenticated + protected route → redirect to /sign-in with returnTo
  if (!session && isProtected) {
    const loginUrl = new URL("/sign-in", request.url);
    loginUrl.searchParams.set("returnTo", pathname);
    return NextResponse.redirect(loginUrl);
  }

  // Already authenticated + auth route → redirect to /dashboard
  if (session && isAuthRoute) {
    return NextResponse.redirect(new URL("/dashboard", request.url));
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!api|_next/static|_next/image|favicon.ico).*)"],
};
```

IMPORTANT — Security note in code comment: Middleware is an optimistic guard only (CVE-2025-29927). Sessions MUST also be validated in Server Components (see Plan 04's app layout). Never rely on middleware as the sole auth layer.

Also update `NEXT_PUBLIC_APP_URL` in `.env.local.example` if not already there.
