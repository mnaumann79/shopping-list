From research — swipe-to-delete (native Pointer Events, touch only):
```typescript
const startX = useRef(0);
const [swipeOffset, setSwipeOffset] = useState(0);

function onPointerDown(e: React.PointerEvent) {
  if (e.pointerType !== "touch") return;
  startX.current = e.clientX;
  e.currentTarget.setPointerCapture(e.pointerId);
}

function onPointerMove(e: React.PointerEvent) {
  const delta = e.clientX - startX.current;
  if (delta < 0) setSwipeOffset(Math.max(delta, -120));
}

function onPointerUp() {
  if (swipeOffset < -80) handleDelete(item.id);
  else setSwipeOffset(0); // snap back
}
```

From research — undo toast pattern (sonner 2.x):
```typescript
function handleDelete(itemId: string) {
  const originalItem = item; // capture before state change
  onDelete(itemId);          // optimistic removal in parent ShoppingList

  let undone = false;
  const timeoutId = setTimeout(async () => {
    if (!undone) {
      await fetch(`/api/items/${itemId}`, { method: "DELETE" });
    }
  }, 5000);

  toast("Item removed", {
    action: {
      label: "Undo",
      onClick: () => {
        undone = true;
        clearTimeout(timeoutId);
        onUpdate(originalItem); // restore in parent — onUpdate can re-add the item
      },
    },
  });
}
```

Note: onUpdate on undo restore — ShoppingList.handleUpdate should also handle re-adding an item that was removed (check if item exists in state, add it back if not).

    function onPointerDown(e: React.PointerEvent) {
      if (e.pointerType !== "touch") return;
      startX.current = e.clientX;
      startY.current = e.clientY;
      trackingSwipe.current = false;
      e.currentTarget.setPointerCapture(e.pointerId);
    }

    function onPointerMove(e: React.PointerEvent) {
      if (e.pointerType !== "touch") return;
      const deltaX = e.clientX - startX.current;
      const deltaY = e.clientY - startY.current;
      if (!trackingSwipe.current) {
        // Only track if horizontal movement dominates
        if (Math.abs(deltaX) < Math.abs(deltaY)) return;
        trackingSwipe.current = true;
      }
      if (deltaX < 0) setSwipeOffset(Math.max(deltaX, -120));
    }

    function onPointerUp() {
      if (swipeOffset < -80) {
        handleDelete(item.id);
      } else {
        setSwipeOffset(0);
      }
      trackingSwipe.current = false;
    }
