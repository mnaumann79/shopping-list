  <action>
    Replace the placeholder dashboard page with the full Server Component implementation.

    New page.tsx:
    ```typescript
    import { auth } from "@/lib/auth";
    import { headers } from "next/headers";
    import { redirect } from "next/navigation";
    import { db } from "@/db";
    import { items, stores, itemStores } from "@/db/schema";
    import { eq, and, desc } from "drizzle-orm";
    import { ensureDefaultCategories, fetchCategories } from "@/lib/categories";
    import { ShoppingList } from "./_components/ShoppingList";

    async function fetchItems(orgId: string) {
      const rawRows = await db
        .select({ item: items, storeId: itemStores.storeId })
        .from(items)
        .leftJoin(itemStores, eq(itemStores.itemId, items.id))
        .where(and(eq(items.organizationId, orgId), eq(items.isActive, true)))
        .orderBy(desc(items.createdAt));

      // Group storeIds by itemId
      const map = new Map<string, { item: typeof rawRows[0]["item"]; storeIds: string[] }>();
      for (const row of rawRows) {
        if (!map.has(row.item.id)) {
          map.set(row.item.id, { item: row.item, storeIds: [] });
        }
        if (row.storeId) {
          map.get(row.item.id)!.storeIds.push(row.storeId);
        }
      }
      return Array.from(map.values()).map(({ item, storeIds }) => ({ ...item, storeIds }));
    }

    async function fetchStores(orgId: string) {
      return db
        .select()
        .from(stores)
        .where(and(eq(stores.organizationId, orgId), eq(stores.isActive, true)))
        .orderBy(stores.name);
    }

    export default async function DashboardPage() {
      const session = await auth.api.getSession({ headers: await headers() });
      if (!session) redirect("/sign-in");

      const orgId = session.session.activeOrganizationId;
      if (!orgId) redirect("/onboarding");

      // Seed defaults idempotently — runs once per household
      await ensureDefaultCategories(orgId);

      const [categories, itemList, storeList] = await Promise.all([
        fetchCategories(orgId),
        fetchItems(orgId),
        fetchStores(orgId),
      ]);

      return (
        <div className="min-h-screen bg-background">
          <header className="border-b px-4 py-3 flex items-center justify-between sticky top-0 bg-background z-10">
            <h1 className="text-base font-semibold">Shopping List</h1>
            <a href="/household/settings" className="text-sm text-muted-foreground hover:underline">
              Settings
            </a>
          </header>
          <main>
            <ShoppingList
              initialItems={itemList}
              categories={categories}
              stores={storeList}
            />
          </main>
        </div>
      );
    }
    ```

    Remove the old HouseholdSwitcher and LogoutButton imports — these are not needed in the new dashboard design (settings link replaces them for navigation). Logout is accessible via Settings.

    Note: The header is simplified for mobile — "Shopping List" title + Settings link. This is consistent with the mobile-first design decision.
