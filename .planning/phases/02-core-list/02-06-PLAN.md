```typescript
const [collapsed, setCollapsed] = useState<Set<string>>(new Set());
function toggleCategory(categoryId: string) {
  setCollapsed(prev => {
    const next = new Set(prev);
    next.has(categoryId) ? next.delete(categoryId) : next.add(categoryId);
    return next;
  });
}
```
