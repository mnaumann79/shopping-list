  <action>
    Create `src/lib/categories.ts`:

    ```typescript
    import { db } from "@/db";
    import { categories } from "@/db/schema";
    import { eq, and } from "drizzle-orm";

    export const DEFAULT_CATEGORIES = [
      "Produce", "Dairy", "Bakery", "Meat", "Frozen",
      "Pantry", "Beverages", "Household", "Personal Care", "Other",
    ] as const;

    export async function ensureDefaultCategories(orgId: string): Promise<void> {
      const rows = DEFAULT_CATEGORIES.map((name) => ({
        organizationId: orgId,
        name,
        isDefault: true,
      }));
      await db.insert(categories).values(rows).onConflictDoNothing();
    }

    export async function fetchCategories(orgId: string) {
      const allCategories = await db
        .select()
        .from(categories)
        .where(and(eq(categories.organizationId, orgId), eq(categories.isActive, true)));

      const defaults = allCategories.filter((c) => c.isDefault);
      const custom = allCategories.filter((c) => !c.isDefault);

      return [
        ...defaults.sort(
          (a, b) =>
            DEFAULT_CATEGORIES.indexOf(a.name as typeof DEFAULT_CATEGORIES[number]) -
            DEFAULT_CATEGORIES.indexOf(b.name as typeof DEFAULT_CATEGORIES[number])
        ),
        ...custom.sort((a, b) => a.name.localeCompare(b.name)),
      ];
    }
    ```

    Note: onConflictDoNothing() works here because plan 02-01 added unique(organizationId, name) to the categories table. If the constraint does not exist in the environment yet (migration not applied), the seed will still work but loses idempotency protection — plan 02-01 must run first.
